function drawHazard(h) { ctx.save(); ctx.shadowBlur = 25; ctx.shadowColor = "#00aaff"; // Blue neon outer block ctx.fillStyle = "#00aaff"; ctx.fillRect( h.x, h.y, h.size, h.size ); // Bright inner block ctx.shadowBlur = 10; ctx.shadowColor = "#7de8ff"; ctx.fillStyle = "#62dcff"; ctx.fillRect( h.x + 5, h.y + 5, h.size - 10, h.size - 10 ); // Dark center ctx.fillStyle = "#10263a"; ctx.fillRect( h.x + h.size * 0.25, h.y + h.size * 0.25, h.size * 0.5, h.size * 0.5 ); ctx.restore(); }
